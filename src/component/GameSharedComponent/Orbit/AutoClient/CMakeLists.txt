set(ORBIT_AUTO_CLIENT_NAME OrbitAutoClient)
set(ORBIT_AUTO_CLIENT_ROOT_DIRECTORY orbit-auto-client)

project_component_declare_service(
  ${ORBIT_AUTO_CLIENT_NAME}
  "${CMAKE_CURRENT_LIST_DIR}"
  OUTPUT_TARGET_NAME
  ORBIT_AUTO_CLIENT_TARGET_NAME
  OUTPUT_NAME
  "orbit-auto-client"
  RUNTIME_OUTPUT_DIRECTORY
  "${ORBIT_AUTO_CLIENT_ROOT_DIRECTORY}/bin"
  USE_COMPONENTS
  "OrbitSDK"
  SOURCES
  "${CMAKE_CURRENT_LIST_DIR}/main.cpp"
  RESOURCE_DIRECTORIES
  "${ORBIT_AUTO_CLIENT_ROOT_DIRECTORY}/log")

target_compile_features(${ORBIT_AUTO_CLIENT_TARGET_NAME} PRIVATE cxx_std_17)
target_compile_definitions(${ORBIT_AUTO_CLIENT_TARGET_NAME}
  PRIVATE GAME_SHARED_COMPONENT_SERVER_MODE=1
          "PROJECT_PROTOCOL_NAMESPACE_ID=${PROJECT_NAMESPACE}")
